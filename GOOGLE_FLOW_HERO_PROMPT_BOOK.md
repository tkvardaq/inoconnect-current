```powershell
node "C:\Users\talha\comfy-mcp\server.mjs" image --prompt "Widescreen 16:9 commercial photography inside an elite digital agency headquarters..." --width 1920 --height 1080 --steps 10 --out "D:\videos for reels\comfy_hero_plate.png"
```
